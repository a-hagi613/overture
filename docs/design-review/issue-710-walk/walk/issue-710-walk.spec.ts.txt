async function shot(page: Page, name: string, dark = false) {
  if (dark) {
    await page.emulateMedia({ colorScheme: "dark" });
  }

  for (const [width, height, suffix] of [
    [390, 844, "390"],
    [1280, 900, "1280"],
  ] as const) {
    await page.setViewportSize({ width, height });
    await page.waitForTimeout(500);
    await page.screenshot({
      path: join(SHOTS, `${name}-${suffix}${dark ? "-dark" : ""}.png`),
      fullPage: true,
    });
  }

  await page.setViewportSize({ width: 1280, height: 900 });
  await page.emulateMedia({ colorScheme: "light" });
}

async function magicLink(email: string) {
  const mail = await (await fetch(`${BASE}/__test__/mail`)).json();
  const message = mail.messages.findLast((m: { to: string }) => m.to === email);

  if (message === undefined) {
    throw new Error(`no mail to ${email}`);
  }

  const link = /https?:\/\/[^\s"']+|\/api\/auth\/[^\s"']+/u.exec(message.body)?.[0];

  return link.startsWith("/") ? `${BASE}${link}` : link;
}

async function fillGaps(page: Page, profile: Profile) {
  for (const [label, value] of Object.entries(profile.gaps)) {
    const field = page.getByLabel(label, { exact: true });

    if ((await field.count()) === 0) {
      continue;
    }

    const current = await field.first().inputValue();

    if (current.trim() !== "") {
      continue;
    }

    const tag = await field.first().evaluate((element) => element.tagName);

    if (tag === "SELECT") {
      await field.first().selectOption(value);
    } else {
      await field.first().fill(value);
    }
  }
}

async function waitForKind(kind: string, timeoutMs: number) {
  const deadline = Date.now() + timeoutMs;

  while (Date.now() < deadline) {
    const { jobs } = (await (await fetch(`${BASE}/__test__/jobs`)).json()) as {
      jobs: { createdAt: number; kind: string; status: string }[];
    };
    const latest = jobs
      .filter((job) => job.kind === kind)
      .toSorted((a, b) => b.createdAt - a.createdAt)[0];

    if (latest !== undefined && (latest.status === "done" || latest.status === "dead")) {
      return latest.status;
    }

    await new Promise((resolve) => {
      setTimeout(resolve, 5000);
    });
  }

  throw new Error(`${kind} did not finish`);
}

function signWebhook(payload: string) {
  const secret = process.env.STRIPE_WEBHOOK_SECRET ?? "";
  const timestamp = Math.floor(Date.now() / 1000);
  const signature = createHmac("sha256", secret)
    .update(`${String(timestamp)}.${payload}`)
    .digest("hex");

  return `t=${String(timestamp)},v1=${signature}`;
}

async function deliverInvoicePaid(sessionId: string) {
  const key = process.env.STRIPE_SECRET_KEY ?? "";
  const auth = { Authorization: `Bearer ${key}` };
  let invoice: unknown = null;

  for (let attempt = 0; attempt < 12; attempt += 1) {
    const session = (await (
      await fetch(
        `https://api.stripe.com/v1/checkout/sessions/${sessionId}?expand[]=subscription.latest_invoice`,
        { headers: auth },
      )
    ).json()) as { subscription?: { latest_invoice?: unknown } };
    invoice = session.subscription?.latest_invoice;

    if (invoice !== undefined && invoice !== null && typeof invoice !== "string") {
      break;
    }

    await new Promise((resolve) => {
      setTimeout(resolve, 5000);
    });
  }

  if (invoice === undefined || invoice === null || typeof invoice === "string") {
    throw new Error("no invoice on the subscription");
  }

  const payload = JSON.stringify({
    api_version: "2026-11-01",
    created: Math.floor(Date.now() / 1000),
    data: { object: invoice },
    id: `evt_walk_${String(Date.now())}`,
    livemode: false,
    object: "event",
    pending_webhooks: 1,
    request: { id: null, idempotency_key: null },
    type: "invoice.paid",
  });
  const response = await fetch(`${BASE}/webhooks/stripe`, {
    body: payload,
    headers: { "content-type": "application/json", "stripe-signature": signWebhook(payload) },
    method: "POST",
  });

  if (!response.ok) {
    throw new Error(`webhook refused ${String(response.status)}`);
  }
}

function seenPromptsFor(profile: Profile) {
  let seen = reviewSeen.get(profile.key);

  if (seen === undefined) {

async function answerReviews(page: Page, profile: Profile) {
  await page.goto(`${BASE}/review`);
  await page.waitForTimeout(2500);
  await shot(page, "review");

  let answered = 0;
  const seen = seenPromptsFor(profile);

  for (let index = 0; index < 12; index += 1) {
    const answer = page.getByRole("button", { name: /^Answer\b/u }).first();

    if ((await answer.count()) === 0) {
      return answered;
    }

    await answer.click();
    const dialog = page.getByRole("dialog");
    await dialog.waitFor({ timeout: 15_000 });

    const sections = await dialog.locator("section").all();
    const prompts = [];

    for (const section of sections) {
      prompts.push((await section.getByRole("heading", { level: 3 }).allTextContents())[0] ?? "");
    }

    if (prompts.some((prompt) => seen.has(prompt))) {
      console.log(`[${profile.key}] question came back unanswered; skipping that application`);
      await dialog.getByRole("button", { name: /Skip this application/u }).click();
      await page.waitForTimeout(1500);

      if ((await page.getByRole("button", { name: /^Answer\b/u }).count()) > 0) {
        await page.goto(`${BASE}/review`);
        await page.waitForTimeout(1500);
      }

      continue;
    }

    for (const prompt of prompts) {
      seen.add(prompt);
    }

    for (const section of sections) {
      const prompt = (await section.getByRole("heading", { level: 3 }).allTextContents())[0] ?? "";
      const wanted = answerFor(profile, prompt);

      for (const textbox of await section.getByRole("textbox").all()) {
        if ((await textbox.inputValue().catch(() => "")) === "") {
          await textbox.fill(wanted);
        }
      }

      for (const group of await section.getByRole("group").all()) {
        const buttons = await group.getByRole("button").all();

        if (buttons.length === 0) {
          continue;
        }

        const matching = group.getByRole("button", {
          name: new RegExp(`^${escapeRegExp(wanted)}$`, "u"),
        });

        if ((await matching.count()) > 0) {
          await matching.first().click();
        } else {
          const no = group.getByRole("button", { name: "No", exact: true });

          if ((await no.count()) > 0) {
            await no.first().click();
          } else {
            await buttons[0]?.click();
          }
        }
      }

      for (const checkbox of await section.locator('input[type="checkbox"]').all()) {
        if (!(await checkbox.isChecked().catch(() => true))) {
          await checkbox.check();
        }
      }
    }

    const save = dialog.getByRole("button", { name: /Save and continue/u });

    if (!(await save.isEnabled().catch(() => false))) {
      console.log(`[${profile.key}] review form save stayed disabled; skipping`);
      await dialog
        .getByRole("button", { name: /Skip this application/u })
        .click()
        .catch(() => {});
      await page.waitForTimeout(1500);

      continue;
    }

    await save.click({ timeout: 15_000 });
    await page.waitForTimeout(2000);
    answered += 1;

    if ((await page.getByRole("button", { name: /^Answer\b/u }).count()) > 0) {
      await page.goto(`${BASE}/review`);
      await page.waitForTimeout(2000);
    }
  }

  return answered;
}

function escapeRegExp(value: string) {
  return value.replaceAll(/[.*+?^${}()|[\]\\]/gu, "\\$&");
}

function answerFor(profile: Profile, prompt: string) {
  const gaps = profile.gaps;

  if (/location \(city\)|city|town/iu.test(prompt)) {
    return gaps.City ?? "Ottawa";
  }

  if (/^state$|^province$|state\/province/iu.test(prompt)) {
    return "ON";
  }

  if (/zip|postal/iu.test(prompt)) {
    return "K1A 0A6";
  }

  if (/^gender$|gender/iu.test(prompt)) {
    return profile.gender ?? "Female";
  }

  if (/^nationality$/iu.test(prompt)) {
    return gaps.Country ?? "Canada";
  }

  if (/country/iu.test(prompt)) {
    return gaps.Country ?? "Canada";
  }

  if (/phone|mobile|telephone/iu.test(prompt)) {
    return gaps.Phone ?? "+1 613 555 0100";
  }

  if (/salary|compensation|expectation/iu.test(prompt)) {
    return gaps["Salary expectation"] ?? "CAD 100,000 a year";
  }

  if (/notice|earliest start|start date/iu.test(prompt)) {
    return gaps["Notice period or earliest start"] ?? "Two weeks";
  }

  if (/relocat/iu.test(prompt)) {
    return gaps["Willing to relocate"] ?? "No";
  }

  if (/sponsor|visa/iu.test(prompt)) {
    return gaps["Needs sponsorship"] ?? "No";
  }

  if (/hear about|how did you (first )?(hear|find)|find this (position|role|job)/iu.test(prompt)) {
    return "LinkedIn Job Search";
  }

  if (/linkedin/iu.test(prompt)) {
    return gaps.LinkedIn ?? "";
  }

  if (/language/iu.test(prompt)) {
    return gaps["Languages and proficiency level"] ?? "English (native)";
  }

  if (/authoriz|allowed to work|legally/iu.test(prompt)) {
    return gaps["Legally allowed to work in the country"] ?? "Yes";
  }

  return "No";
}

    if (await answersPending(page)) {

      continue;
    }

    const applications = await applicationsOf(profile);
    const preparing = applications.filter(

    ).length;

    if (preparing > 0) {
      console.log(`[${profile.key}] waiting for ${String(preparing)} preparations`);
      await page.waitForTimeout(8000);

    await page.goto(`${BASE}/ready`);
    await page.waitForTimeout(1500);

    if ((await page.getByRole("button", { name: "Approve", exact: true }).count()) > 0) {
      return;
    }

    await page.waitForTimeout(5000);
  }
}

async function skipSendSpacing(profile: Profile) {
  for (let round = 0; round < 8; round += 1) {
    const applications = await applicationsOf(profile);
    const waiting = applications.filter(
      (application) =>
        (application.status === "approved" || application.status === "scheduled") &&
        application.scheduledFor !== null,
    );
    const next = waiting.map((application) => application.scheduledFor ?? 0).filter((at) => at > 0);

    if (next.length === 0) {
      await new Promise((resolve) => {
        setTimeout(resolve, 2000);
      });

      continue;
    }

    await advanceClockTo(Math.min(...next) + 2000);
    await new Promise((resolve) => {
      setTimeout(resolve, 4000);
    });
  }
}

  for (const application of applications) {
    const attempts = await attemptsOf(application.id);

    if (attempts.length === 0) {
      continue;
    }

    console.log(
      `[${profile.key}] ${application.companyName} (${application.ats}) status=${application.status} reason=${application.statusReason ?? "-"}`,
    );

    for (const attempt of attempts) {
      console.log(
        `[${profile.key}]   attempt ${String(attempt.n)} outcome=${attempt.outcome ?? "-"} error=${attempt.errorCode ?? "-"} submitClickedAt=${attempt.submitClickedAt === null ? "-" : String(attempt.submitClickedAt)}`,
      );
    }

    summary.push({ application, attempts });

    if (root === undefined) {
      continue;
    }

    const dir = join(root, "screenshots", userId, application.id);

    if (!existsSync(dir)) {
      continue;
    }

    for (const file of readdirSync(dir)) {
      if (!/-(?:before|after)_submit-\d+\.png$/u.test(file)) {
        continue;
      }

      const phase = file.includes("-before_submit-") ? "before" : "after";
      const name = `${profile.key}-send-${application.ats}-${phase}-submit.png`;
      copyFileSync(join(dir, file), join(SHOTS, name));
      console.log(`[${profile.key}]   copied ${name}`);
    }
  }

  return summary;
}

async function planAndTracker(page: Page, profile: Profile) {
  console.log(`[${profile.key}] choose plan`);
  await choosePlan(page, profile.plan);
  console.log(`[${profile.key}] choose picks`);
  const count = profile.plan === "free_trial" ? 3 : 10;
  await choosePicks(page, count);
  console.log(`[${profile.key}] checkout`);
  await shot(page, `${profile.key}-05-matches`, true);

  await page.waitForTimeout(5000);
  console.log("picks selected", await page.getByRole("button", { name: /^Selected /u }).count());

  if (profile.plan === "free_trial") {
    const cta = page.getByRole("button", { name: "Start my 3 free applications" });
    await expect(cta).toBeEnabled({ timeout: 45_000 });
    await cta.click();
    await page.waitForTimeout(1200);
    await page.getByRole("checkbox").check();
    await page.getByRole("button", { name: "Continue for free" }).click();
    await fillStripe(page, false);
    console.log(`[${profile.key}] card returned`);
    await page.waitForURL(/\/tracker/u, { timeout: 120_000 });
  } else {
    const cta = page.getByRole("button", {
      name: profile.plan === "week" ? /Start Weekly/u : /Start Monthly/u,
    });
    await expect(cta).toBeEnabled({ timeout: 45_000 });
    await cta.click();
    await page.waitForTimeout(1200);
    const dialog = page.getByRole("dialog");
    await dialog.waitFor({ timeout: 15_000 });

    if ((await dialog.getByRole("checkbox").count()) > 0) {
      await dialog.getByRole("checkbox").check();
    }

    await dialog
      .getByRole("button", {
        name: profile.plan === "week" ? /Start Weekly/u : /Start Monthly/u,
      })
      .click();
    await page.waitForTimeout(800);
    const session = await fillStripe(page, true);
    await page.waitForURL(/\/tracker/u, { timeout: 120_000 });

    if (session !== null) {
      await deliverInvoicePaid(session);
      await page.goto(`${BASE}/tracker?billing=success&plan=${profile.plan}`);
      await page.waitForTimeout(3000);
    }
  }

  console.log(`[${profile.key}] tracker`);
  await page.waitForTimeout(5000);
  await shot(page, `${profile.key}-06-tracker`);
  await settleWork(page, profile);
  await page.goto(`${BASE}/ready`);
  await page.waitForTimeout(3000);
  await shot(page, `${profile.key}-07-ready`);
  console.log(`[${profile.key}] approvals`);

  for (let round = 0; round < 3; round += 1) {
    await approveEverything(page, profile);
    await shot(page, `${profile.key}-08-scheduled`);
    console.log(`[${profile.key}] waiting for sends (round ${String(round)})`);

    let sawAttempt = false;

    const sendDeadline = Date.now() + 8 * 60_000;

    while (Date.now() < sendDeadline) {
      await skipSendSpacing(profile);
      await page.waitForTimeout(3000);

      if (await answersPending(page)) {
        break;
      }

      const applications = await applicationsOf(profile);
      const busy = applications.filter(
        (application) =>
          application.status === "scheduled" ||
          application.status === "sending" ||
          application.status === "preparing",
      );

      if (busy.length === 0) {
        break;
      }

      const rows = await sendActivity(profile);

      if (rows.length > 0) {
        sawAttempt = true;
      }
    }

    if (await answersPending(page)) {
      await settleWork(page, profile);

      continue;
    }

    if (sawAttempt) {
      break;
    }
  }

  await captureSendEvidence(profile);
  await page.goto(`${BASE}/tracker`);
  await page.waitForTimeout(3000);
  await shot(page, `${profile.key}-09-tracker-final`);
}
